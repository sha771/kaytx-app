/**
 * Consolidated Workflow Service
 * Merges workflow-engine, workflow-automation, and unified-workflow functionality
 * Eliminates code duplication and provides a single source of truth for workflow management

// Unified interfaces combining the best of all three services
export interface WorkflowDefinition {
  id: string;
  name: string;
  description?: string;
  organizationId: string;
  category: 'marketing' | 'sales' | 'support' | 'operations' | 'hr' | 'finance' | 'custom';
  status: 'active' | 'inactive' | 'draft' | 'archived';
  triggers: WorkflowTrigger[];
  steps: WorkflowStep[];
  variables: WorkflowVariable[];
  settings: WorkflowSettings;
  version: number;
  createdBy?: string;
  createdAt: Date;
  updatedAt: Date;
  metadata?: Record<string, any>;
}
 
export interface WorkflowTrigger {
  id: string;
  type: 'manual' | 'scheduled' | 'event' | 'webhook' | 'data_change';
  config: Record<string, any>;
  conditions?: Record<string, any>;
  isActive: boolean;
}
 
export interface WorkflowStep {
  id: string;
  name: string;
  type: 'action' | 'condition' | 'delay' | 'parallel' | 'loop' | 'notification' | 'approval' | 'webhook' | 'sub_workflow';
  config: Record<string, any>;
  position: { x: number; y: number };
  connections: Array<{ from: string; to: string; condition?: string }>;
  nextStepId?: string;
  errorStepId?: string;
  timeout?: number;
  retryPolicy?: {
    maxAttempts: number;
    backoffType: 'fixed' | 'exponential' | 'linear';
    backoffMultiplier?: number;
    delay?: number;
  };
}
 
export interface WorkflowVariable {
  id: string;
  name: string;
  type: 'string' | 'number' | 'boolean' | 'date' | 'object' | 'array';
  defaultValue?: any;
  isRequired: boolean;
  description?: string;
}

    allowedRoles?: string[];
    requireApproval?: boolean;

export interface WorkflowExecution {

  context: Record<string, any>;

  parameters: Array<{
    name: string;
    type: string;
    required: boolean;
    description?: string;
  }>;
}

  /**
   * Create a new workflow
   */
  async createWorkflow(
    organizationId: string,
    workflowData: Omit<WorkflowDefinition, 'id' | 'organizationId' | 'createdAt' | 'updatedAt' | 'version'>
  ): Promise<WorkflowDefinition> {
    try {
      const workflowId = crypto.randomUUID();
      const now = new Date();
 
      const workflow: WorkflowDefinition = {
        ...workflowData,
        id: workflowId,
        organizationId,
        createdAt: now,
        updatedAt: now,
        version: 1
      };
 
      // Save to database
      await pgDb.insert(workflows).values({
        id: workflow.id,
        name: workflow.name,
        description: workflow.description,
        organizationId: workflow.organizationId,
        category: workflow.category,
        status: workflow.status,
        triggers: JSON.stringify(workflow.triggers),
        steps: JSON.stringify(workflow.steps),
        variables: JSON.stringify(workflow.variables),
        settings: JSON.stringify(workflow.settings),
        version: workflow.version,

        metadata: JSON.stringify(workflow.metadata || {}),

    }
  }
 
  /**
   * Execute a workflow
   */
  async executeWorkflow(
    workflowId: string,
    organizationId: string,
    triggerData: Record<string, any> = {},
    userId?: string
  ): Promise<WorkflowExecution> {
    try {
      // Get workflow definition
      const workflow = await this.getWorkflow(workflowId, organizationId);
      if (!workflow) {
        throw new Error('Workflow not found');
      }
 
      if (workflow.status !== 'active') {
        throw new Error('Workflow is not active');
      }
 
      // Check security permissions
      if (workflow.settings.security.allowedUsers && 
          !workflow.settings.security.allowedUsers.includes(userId || '')) {
        throw new Error('Access denied: User not allowed to execute this workflow');
      }
 
      // Create execution record
      const executionId = crypto.randomUUID();
      const execution: WorkflowExecution = {
        id: executionId,
        workflowId,
        organizationId,
        status: 'pending',
        startedAt: new Date(),
        timeoutAt: new Date(Date.now() + workflow.settings.timeout),
        context: this.initializeVariables(workflow.variables, triggerData),
        triggerData,
        completedSteps: [],
        metadata: { userId }
      };
 
      // Save execution to database
      await pgDb.insert(workflowExecutions).values({
        id: execution.id,
        workflowId: execution.workflowId,
        organizationId: execution.organizationId,
        status: execution.status,
        startedAt: execution.startedAt,
        timeoutAt: execution.timeoutAt,
        context: JSON.stringify(execution.context),
        triggerData: JSON.stringify(execution.triggerData),
        completedSteps: JSON.stringify(execution.completedSteps),
        metadata: JSON.stringify(execution.metadata || {})
      });
 
      // Track active execution
      this.activeExecutions.set(executionId, execution);
 
      // Start execution
      this.emit('workflow:started', { execution });
      setImmediate(() => this.runWorkflowExecution(workflow, execution));
 
      return execution;
    } catch (error) {
      throw new Error(`Failed to execute workflow: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }
 
  /**
   * Run workflow execution
   */
  private async runWorkflowExecution(workflow: WorkflowDefinition, execution: WorkflowExecution): Promise<void> {
    try {
      execution.status = 'running';
      await this.updateExecutionStatus(execution.id, 'running');
 
      // Find starting step
      const startStep = this.findStartStep(workflow);
      if (!startStep) {
        throw new Error('No starting step found in workflow');
      }
 
      await this.executeStep(execution.id, startStep.id);
 
    } catch (error) {
      execution.status = 'failed';
      execution.error = error instanceof Error ? error.message : 'Unknown error';
      execution.completedAt = new Date();
      await this.updateExecutionStatus(execution.id, 'failed', execution.error);
      this.emit('workflow:failed', { execution, error });
    } finally {
      this.activeExecutions.delete(execution.id);
    }
  }
 
  /**
   * Find the starting step in a workflow
   */
  private findStartStep(workflow: WorkflowDefinition): WorkflowStep | null {
    const stepIds = new Set(workflow.steps.map(step => step.id));
    const targetStepIds = new Set(
      workflow.steps.flatMap(step => step.connections.map(conn => conn.to))
    );
 
    // Start step is one that has no incoming connections
    const startStepId = Array.from(stepIds).find(id => !targetStepIds.has(id));
    return startStepId ? workflow.steps.find(step => step.id === startStepId) || null : null;
  }
 
  /**
   * Execute a single step
   */
  private async executeStep(executionId: string, stepId: string): Promise<void> {
    const execution = this.activeExecutions.get(executionId);
    if (!execution) return;
 
    const workflow = await this.getWorkflow(execution.workflowId, execution.organizationId);
    if (!workflow) {
      execution.status = 'failed';
      execution.error = 'Workflow not found';
      return;
    }
 
    const step = workflow.steps.find(s => s.id === stepId);
    if (!step) {
      throw new Error(`Step not found: ${stepId}`);
    }
 
    execution.currentStepId = stepId;
 
    try {
      let result: any;
 
      switch (step.type) {
        case 'action':
          result = await this.executeActionStep(execution, step);
          break;
        case 'condition':
          result = await this.executeConditionStep(execution, step);
          break;
        case 'delay':
          result = await this.executeDelayStep(execution, step);
          break;
        case 'parallel':
          result = await this.executeParallelStep(execution, step);
          break;
        case 'notification':
          result = await this.executeNotificationStep(execution, step);
          break;
        case 'approval':
          result = await this.executeApprovalStep(execution, step);
          break;
        case 'webhook':
          result = await this.executeWebhookStep(execution, step);
          break;
        case 'sub_workflow':
          result = await this.executeSubWorkflowStep(execution, step);
          break;
        default:
          throw new Error(`Unsupported step type: ${step.type}`);
      }
 
      execution.completedSteps.push(stepId);
      execution.context[stepId] = result;
 
      // Find next steps
      const nextSteps = this.findNextSteps(workflow, stepId, result);
      
      if (nextSteps.length === 0) {
        // Workflow completed
        execution.status = 'completed';
        execution.completedAt = new Date();
        await this.updateExecutionStatus(execution.id, 'completed');
        this.emit('workflow:completed', execution);
      } else if (nextSteps.length === 1) {
        // Single next step
        await this.executeStep(executionId, nextSteps[0]);
      } else {
        // Multiple next steps (parallel execution)
        await Promise.all(nextSteps.map(nextStepId => 
          this.executeStep(executionId, nextStepId)
        ));
      }
 
    } catch (error) {
      // Check retry policy
      if (step.retryPolicy) {
        const retryCount = execution.retryCounts?.get(stepId) || 0;
        const maxRetries = step.retryPolicy.maxAttempts || 3;
        
        if (retryCount < maxRetries) {
          // Increment retry count
          if (!execution.retryCounts) {
            execution.retryCounts = new Map();
          }
          execution.retryCounts.set(stepId, retryCount + 1);
          
          // Calculate delay
          let delay = step.retryPolicy.delay || 5000;
          if (step.retryPolicy.backoffType === 'exponential') {
            delay *= Math.pow(step.retryPolicy.backoffMultiplier || 2, retryCount);
          } else if (step.retryPolicy.backoffType === 'linear') {
            delay *= (retryCount + 1);
          }
          
          console.warn(`[ConsolidatedWorkflowService] Step ${stepId} failed, retrying (${retryCount + 1}/${maxRetries}) in ${delay}ms`);
          
          // Wait before retry
          await new Promise(resolve => setTimeout(resolve, delay));
          
          // Retry the step
          return await this.executeStep(executionId, stepId);
        } else {
          console.error(`[ConsolidatedWorkflowService] Step ${stepId} failed after ${maxRetries} retries`);
        }
      }
 
      // Check if error step is defined
      if (step.errorStepId) {
        await this.executeStep(executionId, step.errorStepId);
      } else {
        execution.status = 'failed';
        execution.error = error instanceof Error ? error.message : 'Unknown error';
        execution.completedAt = new Date();
        await this.updateExecutionStatus(execution.id, 'failed', execution.error);
        this.emit('workflow:failed', { execution, error });
      }
    }
  }

  /**
   * Evaluate expression
   */
  private evaluateExpression(expression: string, variables: Record<string, any>): boolean {
    try {
      // Replace variables in expression
      let evalExpression = expression;
      for (const [key, value] of Object.entries(variables)) {
        evalExpression = evalExpression.replace(new RegExp(`\\b${key}\\b`, 'g'), JSON.stringify(value));
      }
 
      // Evaluate the expression (be careful with eval in production)
      return Boolean(eval(evalExpression));
    } catch (error) {
      console.error('[ConsolidatedWorkflowService] Expression evaluation failed:', error);
      return false;
    }
  }
 
  /**

  private findNextSteps(workflow: WorkflowDefinition, currentStepId: string, result: any): string[] {
    const currentStep = workflow.steps.find(s => s.id === currentStepId);
    if (!currentStep) return [];
 
    const connections = currentStep.connections.filter(conn => conn.from === currentStepId);
    
    return connections.map(conn => {
      // Check if connection has condition

    }).filter(Boolean) as string[];
  }
