import { db } from '../db/connection';
import { agentMemories, memoryAssociations, memorySummarizationJobs, memoryArchivalJobs, memoryContextWindows } from '../db/drizzle-schema';
import { eq, and, gte, lte, desc, sql, ilike, inArray } from 'drizzle-orm';
import crypto from 'crypto';
import { OpenAI } from 'openai';
 
 
import { createLogger } from '../lib/production-logger';
 
const logger = createLogger(__filename.split('/').pop()?.replace('.ts', '') || 'Service');
 
export interface MemoryData {
  organizationId: string;
  userId?: string;
  agentId?: string;
  sessionId?: string;
  type: 'conversation' | 'knowledge' | 'preference' | 'context';
  content: string;
  importance: number; // 1-10
  priority: 'low' | 'medium' | 'high' | 'critical';
  tags?: string[];
  metadata?: Record<string, any>;
}
 
export interface MemoryFilters {
  organizationId: string;
  userId?: string;
  agentId?: string;
  sessionId?: string;
  type?: string;
  tags?: string[];
  dateRange?: { start: Date; end: Date };
  limit?: number;
  offset?: number;
  semanticSearch?: boolean;
  query?: string;
  minImportance?: number;
}
 
export interface MemoryStats {
  totalMemories: number;
  memoriesByType: Record<string, number>;
  averageImportance: number;
  memoriesByPriority: Record<string, number>;
  topTags: Array<{ tag: string; count: number }>;
  memoriesByDate: Array<{ date: string; count: number }>;
}
 
export class UnifiedMemoryService {
  private openai: OpenAI;
 
  constructor() {
    this.openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY,
    });
  }
 
  async storeMemory(memoryData: MemoryData): Promise<any> {
    try {
      const memoryId = crypto.randomUUID();
      const now = new Date();
      
      // Generate embedding for semantic search
      const embedding = await this.generateEmbedding(memoryData.content);
      
      // Store the memory
      const [memory] = await db.insert(agentMemories).values({
        id: memoryId,
        organizationId: memoryData.organizationId,
        userId: memoryData.userId,
        agentId: memoryData.agentId,
        sessionId: memoryData.sessionId,
        type: memoryData.type,
        content: memoryData.content,
        importance: memoryData.importance,
        priority: memoryData.priority,
        tags: memoryData.tags || [],
        metadata: memoryData.metadata || {},
        embedding,
        createdAt: now,
        updatedAt: now,
      }).returning();
 
      return memory;
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to store memory:', error);
      throw new Error('Failed to store memory');
    }
  }
 
  async retrieveMemories(filters: MemoryFilters): Promise<any[]> {
    try {
      let query = db.select().from(agentMemories).where(
        eq(agentMemories.organizationId, filters.organizationId)
      );
 
      // Apply filters
      if (filters.userId) {
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          eq(agentMemories.userId, filters.userId)
        ));
      }
 
      if (filters.agentId) {
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          eq(agentMemories.agentId, filters.agentId)
        ));
      }
 
      if (filters.type) {
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          eq(agentMemories.type, filters.type as any)
        ));
      }
 
      if (filters.minImportance) {
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          gte(agentMemories.importance, filters.minImportance)
        ));
      }
 
      if (filters.dateRange) {
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          gte(agentMemories.createdAt, filters.dateRange.start),
          lte(agentMemories.createdAt, filters.dateRange.end)
        ));
      }
 
      if (filters.tags && filters.tags.length > 0) {
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          sql`${agentMemories.tags} && ${filters.tags}`
        ));
      }
 
      // Apply semantic search if requested
      if (filters.semanticSearch && filters.query) {
        const queryEmbedding = await this.generateEmbedding(filters.query);
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          sql`cosine_similarity(${agentMemories.embedding}, ${queryEmbedding}) > 0.7`
        ));
      }
 
      // Apply text search if query provided but not semantic search
      if (filters.query && !filters.semanticSearch) {
        query = query.where(and(
          eq(agentMemories.organizationId, filters.organizationId),
          ilike(agentMemories.content, `%${filters.query}%`)
        ));
      }
 
      // Apply ordering and pagination
      query = query
        .orderBy(desc(agentMemories.createdAt))
        .limit(filters.limit || 20)
        .offset(filters.offset || 0);
 
      const memories = await query;
 
      // Calculate similarity scores for semantic search
      if (filters.semanticSearch && filters.query) {
        const queryEmbedding = await this.generateEmbedding(filters.query);
        for (const memory of memories) {
          memory.similarity = await this.calculateSimilarity(queryEmbedding, memory.embedding);
        }
        memories.sort((a, b) => (b.similarity || 0) - (a.similarity || 0));
      }
 
      return memories;
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to retrieve memories:', error);
      throw new Error('Failed to retrieve memories');
    }
  }
 
  async updateMemory(memoryId: string, updates: Partial<MemoryData>): Promise<any> {
    try {
      const updateData: any = { ...updates, updatedAt: new Date() };
      
      // Generate new embedding if content was updated
      if (updates.content) {
        updateData.embedding = await this.generateEmbedding(updates.content);
      }
 
      const [memory] = await db
        .update(agentMemories)
        .set(updateData)
        .where(eq(agentMemories.id, memoryId))
        .returning();
 
      return memory;
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to update memory:', error);
      throw new Error('Failed to update memory');
    }
  }
 
  async deleteMemory(memoryId: string): Promise<boolean> {
    try {
      const result = await db
        .delete(agentMemories)
        .where(eq(agentMemories.id, memoryId));
 
      return result.rowCount > 0;
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to delete memory:', error);
      throw new Error('Failed to delete memory');
    }
  }
 
  async getMemoryStats(organizationId: string): Promise<MemoryStats> {
    try {
      // Get total memories
      const [totalResult] = await db
        .select({ count: sql<number>`count(*)`.mapWith(Number) })
        .from(agentMemories)
        .where(eq(agentMemories.organizationId, organizationId));
 
      // Get memories by type
      const typeResults = await db
        .select({
          type: agentMemories.type,
          count: sql<number>`count(*)`.mapWith(Number)
        })
        .from(agentMemories)
        .where(eq(agentMemories.organizationId, organizationId))
        .groupBy(agentMemories.type);
 
      // Get memories by priority
      const priorityResults = await db
        .select({
          priority: agentMemories.priority,
          count: sql<number>`count(*)`.mapWith(Number)
        })
        .from(agentMemories)
        .where(eq(agentMemories.organizationId, organizationId))
        .groupBy(agentMemories.priority);
 
      // Get average importance
      const [avgResult] = await db
        .select({ avg: sql<number>`avg(importance)`.mapWith(Number) })
        .from(agentMemories)
        .where(eq(agentMemories.organizationId, organizationId));
 
      // Get top tags
      const tagResults = await db
        .select({ tags: agentMemories.tags })
        .from(agentMemories)
        .where(eq(agentMemories.organizationId, organizationId));
 
      const tagCounts: Record<string, number> = {};
      tagResults.forEach(result => {
        if (result.tags && Array.isArray(result.tags)) {
          result.tags.forEach(tag => {
            tagCounts[tag] = (tagCounts[tag] || 0) + 1;
          });
        }
      });
 
      const topTags = Object.entries(tagCounts)
        .sort(([, a], [, b]) => b - a)
        .slice(0, 10)
        .map(([tag, count]) => ({ tag, count }));
 
      // Get memories by date
      const dateResults = await db
        .select({
          date: sql<string>`date(created_at)`.as('date'),
          count: sql<number>`count(*)`.mapWith(Number)
        })
        .from(agentMemories)
        .where(eq(agentMemories.organizationId, organizationId))
        .groupBy(sql`date(created_at)`)
        .orderBy(desc(sql`date(created_at)`));
 
      const memoriesByType = typeResults.reduce((acc, result) => {
        acc[result.type] = result.count;
        return acc;
      }, {} as Record<string, number>);
 
      const memoriesByPriority = priorityResults.reduce((acc, result) => {
        acc[result.priority] = result.count;
        return acc;
      }, {} as Record<string, number>);
 
      return {
        totalMemories: totalResult.count,
        memoriesByType,
        averageImportance: avgResult.avg || 0,
        memoriesByPriority,
        topTags,
        memoriesByDate: dateResults,
      };
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to get memory stats:', error);
      throw new Error('Failed to get memory stats');
    }
  }
 
  private async generateEmbedding(text: string): Promise<number[]> {
    try {
      const response = await this.openai.embeddings.create({
        model: 'text-embedding-ada-002',
        input: text,
      });
 
      return response.data[0].embedding;
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to generate embedding:', error);
      throw new Error('Failed to generate embedding');
    }
  }
 
  private async calculateSimilarity(embedding1: number[], embedding2: number[]): Promise<number> {
    if (embedding1.length !== embedding2.length) {
      return 0;
    }
 
    let dotProduct = 0;
    let norm1 = 0;
    let norm2 = 0;
 
    for (let i = 0; i < embedding1.length; i++) {
      dotProduct += embedding1[i] * embedding2[i];
      norm1 += embedding1[i] * embedding1[i];
      norm2 += embedding2[i] * embedding2[i];
    }
 
    norm1 = Math.sqrt(norm1);
    norm2 = Math.sqrt(norm2);
 
    if (norm1 === 0 || norm2 === 0) {
      return 0;
    }
 
    return dotProduct / (norm1 * norm2);
  }
 
  async searchSimilarMemories(query: string, organizationId: string, limit: number = 10): Promise<any[]> {
    try {
      const queryEmbedding = await this.generateEmbedding(query);
      
      const memories = await db
        .select()
        .from(agentMemories)
        .where(eq(agentMemories.organizationId, organizationId))
        .orderBy(sql`cosine_similarity(${agentMemories.embedding}, ${queryEmbedding}) DESC`)
        .limit(limit);
 
      // Calculate similarity scores
      for (const memory of memories) {
        memory.similarity = await this.calculateSimilarity(queryEmbedding, memory.embedding);
      }
 
      return memories.filter(memory => memory.similarity > 0.7);
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to search similar memories:', error);
      throw new Error('Failed to search similar memories');
    }
  }
 
  async createMemoryAssociation(memoryId1: string, memoryId2: string, type: string, strength: number): Promise<any> {
    try {
      const [association] = await db.insert(memoryAssociations).values({
        id: crypto.randomUUID(),
        memoryId1,
        memoryId2,
        type,
        strength,
        createdAt: new Date(),
      }).returning();
 
      return association;
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to create memory association:', error);
      throw new Error('Failed to create memory association');
    }
  }
 
  async getRelatedMemories(memoryId: string, organizationId: string): Promise<any[]> {
    try {
      const associations = await db
        .select()
        .from(memoryAssociations)
        .where(
          sql`memory_id1 = ${memoryId} OR memory_id2 = ${memoryId}`
        );
 
      const relatedMemoryIds = associations.map(assoc => 
        assoc.memoryId1 === memoryId ? assoc.memoryId2 : assoc.memoryId1
      );
 
      if (relatedMemoryIds.length === 0) {
        return [];
      }
 
      const relatedMemories = await db
        .select()
        .from(agentMemories)
        .where(
          and(
            eq(agentMemories.organizationId, organizationId),
            inArray(agentMemories.id, relatedMemoryIds)
          )
        );
 
      return relatedMemories;
    } catch (error) {
      logger.error('[UnifiedMemoryService] Failed to get related memories:', error);
      throw new Error('Failed to get related memories');
    }
  }
}
